Reinforcement:-

import numpy as np 
 
m_states = 5 
actions = [0, 1] 
 
rewards = [-1, -1, -1, -1, 10] 
goal_state = 4 
 
alpha = 0.1 
gama = 0.9 
epsilon = 0.1 
 
q_table = np.zeros((m_states, len(actions))) 
 
n_episodes = 1000 
 
for episode in range(n_episodes): 
    state = 0 
 
    while state != goal_state: 
 
        # Choose action using epsilon-greedy policy 
        if np.random.uniform(0, 1) < epsilon: 
            action = np.random.choice(actions) 
        else: 
            action = np.argmax(q_table[state]) 
 
        # Take action 
        new_state = state + 1 if action == 1 else max(0, state - 1) 
 
        # Get reward 
        reward = rewards[new_state] 
 
        # Q-learning update 
        q_table[state, action] += alpha * ( 
            reward 
            + gama * np.max(q_table[new_state]) 
            - q_table[state, action] 
        ) 
 
        # Move to the new state 
        state = new_state 
 
 
# Find optimal path after training 
state = 0 
path = [state] 
 
while state != goal_state: 
    action = np.argmax(q_table[state]) 
    state = state + 1 if action == 1 else max(0, state - 1) 
    path.append(state) 

print("Optimal path from start to goal:", path) 
print("\nQ-table:") 
print(q_table)
